! ============================================================
! RTR-CORE - Cisco 2911 - Inter-VLAN routing, DHCP, ACLs, OSPF
! Project 300 - Networking
!
! HOW TO USE
!   1. Open the device CLI, then: enable
!   2. configure terminal
!   3. Paste everything between BEGIN and END
!   4. Generate the SSH key and finish line config (end of file)
!
! The ACLs are defined BEFORE the interfaces so the
! "ip access-group" lines work.
!
! NOTE: Passwords are placeholders (Cisco123!). Change them.
! ============================================================
! ---------------------- BEGIN ----------------------
hostname RTR-CORE
no ip domain-lookup
ip domain-name bank.local
enable secret Cisco123!
username admin privilege 15 secret Cisco123!
!
! ---- ACL: Staff VLAN (inbound on Gi0/1.20) ----
ip access-list extended STAFF-IN
 permit udp any eq bootpc any eq bootps
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.10.10 eq 1433
 permit udp 192.168.20.0 0.0.0.255 host 192.168.10.10 eq 53
 permit tcp 192.168.20.0 0.0.0.255 host 192.168.10.10 eq 53
 deny ip 192.168.20.0 0.0.0.255 192.168.0.0 0.0.255.255
 deny ip 192.168.20.0 0.0.0.255 10.0.0.0 0.255.255.255
 permit ip any any
!
! ---- ACL: Servers VLAN (inbound on Gi0/1.10) ----
ip access-list extended SERVERS-IN
 permit udp host 192.168.10.10 eq 53 192.168.20.0 0.0.0.255
 permit udp host 192.168.10.10 eq 1812 host 2.2.2.2
 permit udp host 192.168.10.10 eq 1813 host 2.2.2.2
 permit udp host 192.168.10.10 eq 1812 192.168.30.0 0.0.0.255
 permit udp host 192.168.10.10 eq 1813 192.168.30.0 0.0.0.255
 permit icmp 192.168.10.0 0.0.0.255 192.168.30.0 0.0.0.255 echo-reply
 permit tcp 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255 established
 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip any any
!
! ---- DHCP for the Staff VLAN ----
ip dhcp excluded-address 192.168.20.1 192.168.20.20
ip dhcp pool STAFF
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.10.10
!
! ---- Interfaces ----
interface loopback0
 ip address 2.2.2.2 255.255.255.255
!
interface gi0/0
 description To RTR-EDGE
 ip address 10.0.1.2 255.255.255.252
 ip ospf network point-to-point
 no shutdown
!
interface gi0/1
 description Trunk to SW-01
 no ip address
 no shutdown
!
interface gi0/1.10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip access-group SERVERS-IN in
!
interface gi0/1.20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip access-group STAFF-IN in
!
interface gi0/1.30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
!
! ---- OSPF (area 0) ----
router ospf 1
 router-id 2.2.2.2
 passive-interface default
 no passive-interface gi0/0
 network 10.0.1.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 2.2.2.2 0.0.0.0 area 0
!
end
! ----------------------- END -----------------------
!
! SSH SETUP (run by hand after the paste above)
!   configure terminal
!   crypto key generate rsa      <- Packet Tracer: enter 2048 when asked
!   ip ssh version 2
!   line con 0
!    logging synchronous
!   line vty 0 15
!    transport input ssh
!    login local
!   end
!   write memory
!
! If the deny lines are missing from "show access-lists",
! insert them (they must sit BEFORE "permit ip any any"):
!   ip access-list extended STAFF-IN
!    45 deny ip 192.168.20.0 0.0.0.255 192.168.0.0 0.0.255.255
!    46 deny ip 192.168.20.0 0.0.0.255 10.0.0.0 0.255.255.255
!   ip access-list extended SERVERS-IN
!    75 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255
!
! VERIFY
!   show ip interface brief
!   show ip ospf neighbor
!   show ip route
!   show access-lists
!   show ip dhcp binding
