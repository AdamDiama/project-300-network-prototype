! ------------------------------------------------------------
! 1. SW-01: server port becomes a trunk (tagged VMs)
!    Replaces the Packet Tracer access-port config on fa0/1.
!    The hypervisor must tag each VM into its VLAN,
!    or the server needs multiple NICs.
! ------------------------------------------------------------
interface fa0/1
 description Server host (tagged VMs)
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,30
 switchport nonegotiate


! ------------------------------------------------------------
! 2. SSH keys: one-line form works on real hardware
! ------------------------------------------------------------
crypto key generate rsa modulus 2048


! ------------------------------------------------------------
! 3. OSPF authentication (both routers, same key on each side)
!    Removed in Packet Tracer because it was inconsistent.
! ------------------------------------------------------------
! RTR-EDGE:
interface gi0/1
 ip ospf message-digest-key 1 md5 <strong-key>
router ospf 1
 area 0 authentication message-digest

! RTR-CORE:
interface gi0/0
 ip ospf message-digest-key 1 md5 <strong-key>
router ospf 1
 area 0 authentication message-digest


! ------------------------------------------------------------
! 4. ACL logging on RTR-CORE
!    Add "log" to the deny lines to record blocked attempts.
!    Rebuild the ACL entries with the log keyword, for example:
! ------------------------------------------------------------
ip access-list extended STAFF-IN
 no 45
 no 46
 45 deny ip 192.168.20.0 0.0.0.255 192.168.0.0 0.0.255.255 log
 46 deny ip 192.168.20.0 0.0.0.255 10.0.0.0 0.255.255.255 log
ip access-list extended SERVERS-IN
 no 75
 75 deny ip 192.168.10.0 0.0.0.255 192.168.0.0 0.0.255.255 log
! (If your ACLs were pasted without sequence numbers, run
!  "show access-lists" to see the actual line numbers first.)


! ------------------------------------------------------------
