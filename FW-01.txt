! ============================================================
! FW-01 - Cisco ASA 5506-X - Perimeter firewall, NAT/PAT
! Project 300 - Networking
!
! HOW TO USE
!   1. Open the device CLI, then: enable
!      (the enable password is blank on a new ASA: press Enter)
!   2. configure terminal
!   3. Paste everything between BEGIN and END
!
! If Packet Tracer rejects the policy-map lines, skip them:
! web traffic still works, ICMP replies may not return.
!
! NOTE: The password is a placeholder (Cisco123!). Change it.
! ============================================================
! ---------------------- BEGIN ----------------------
hostname FW-01
enable password Cisco123!
!
interface gi1/1
 nameif outside
 security-level 0
 ip address 203.0.113.2 255.255.255.248
 no shutdown
!
interface gi1/2
 nameif inside
 security-level 100
 ip address 10.0.2.2 255.255.255.252
 no shutdown
!
! Routes
route outside 0.0.0.0 0.0.0.0 203.0.113.1
route inside 192.168.0.0 255.255.0.0 10.0.2.1
route inside 10.0.1.0 255.255.255.252 10.0.2.1
!
! PAT: inside hosts share the outside interface address
object network INSIDE-NET
 subnet 192.168.0.0 255.255.0.0
 nat (inside,outside) dynamic interface
!
! Outbound policy: DNS, HTTP, HTTPS, ICMP only
access-list INSIDE-OUT extended permit udp any any eq 53
access-list INSIDE-OUT extended permit tcp any any eq 80
access-list INSIDE-OUT extended permit tcp any any eq 443
access-list INSIDE-OUT extended permit icmp any any
access-group INSIDE-OUT in interface inside
!
policy-map global_policy
 class inspection_default
  inspect icmp
! ----------------------- END -----------------------
!
! Inbound traffic from outside is denied by default
! (security-level 0 to 100, no permit rules).
!
! VERIFY
!   show interface ip brief
!   show route
!   show xlate                   (after traffic has passed through)
!
! STATUS: configured, but outbound NAT was NOT tested in the
! prototype because there is no external device on Gi1/1.
