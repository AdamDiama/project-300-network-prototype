! ============================================================
! SW-01 - Catalyst 2960 (IOS 15) - Layer 2 access switch
! Project 300 - Networking
!
! HOW TO USE
!   1. Open the device CLI, then: enable
!   2. configure terminal
!   3. Paste everything below the "BEGIN" line
!   4. Generate the SSH key (see the end of this file)
!
! NOTE: Passwords are placeholders (Cisco123!). Change them.
! ============================================================
! ---------------------- BEGIN ----------------------
hostname SW-01
no ip domain-lookup
ip domain-name bank.local
service password-encryption
enable secret Cisco123!
username admin privilege 15 secret Cisco123!
banner motd # Authorised access only #
spanning-tree mode rapid-pvst
!
vlan 10
 name Servers
vlan 20
 name Staff
vlan 30
 name Management
vlan 99
 name Native-Unused
!
interface gi0/1
 description Trunk to RTR-CORE
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,99
 switchport nonegotiate
!
! Packet Tracer version: server on an access port.
! For the physical rack, see physical-rack-changes.txt
interface fa0/1
 description SRV-HOST
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
!
interface fa0/2
 description LAPTOP Staff
 switchport mode access
 switchport access vlan 20
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
!
interface fa0/3
 description LAPTOP Management
 switchport mode access
 switchport access vlan 30
 switchport port-security
 switchport port-security maximum 2
 switchport port-security violation shutdown
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
!
interface range fa0/4 - 24
 description Unused
 switchport mode access
 switchport access vlan 99
 shutdown
!
interface vlan 30
 ip address 192.168.30.3 255.255.255.0
 no shutdown
ip default-gateway 192.168.30.1
!
ip ssh version 2
line con 0
 logging synchronous
line vty 0 15
 transport input ssh
 login local
! ----------------------- END -----------------------
!
! AFTER PASTING (run these by hand, they are not part of the paste):
!   exit                         <- leave line config mode first
!   crypto key generate rsa      <- Packet Tracer: enter 2048 when asked
!   end
!   write memory
!
! On real hardware: crypto key generate rsa modulus 2048
!
! VERIFY
!   show vlan brief
!   show interfaces trunk
!   show port-security interface fa0/2
!   show ip ssh
